    static string[] FindWords(string[] words)
    {
        string[] rows = { "qwertyuiop", "asdfghjkl", "zxcvbnm" };
        Dictionary<char, int> charRow = new Dictionary<char, int>();


        for (int i = 0; i < rows.Length; i++) // Dictionnary of   "qwertyuiop" characters  with --> 0 value
        {                                     // "asdfghjkl" characters  with --> 1 value
            foreach (char c in rows[i])       // "zxcvbnm" characters  with --> 2 value
            {
                charRow[c] = i;
            }
        }


        List<string> result = new List<string>();


        foreach (string word in words)
        {
            int row = charRow[char.ToLower(word[0])]; // the row of the first leter in the word
            bool isValid = true;


            foreach (char c in word)
            {
                if (charRow[char.ToLower(c)] != row) // if one letter in word is not in the same row -->
                                                     // is invalid -->
                                                     // i break it because all word isn t in the same qwerty.
                {
                    isValid = false;
                    break;
                }
            }


            if (isValid)
                result.Add(word);
        }


        return result.ToArray();
    }
