
using System;
using System.Collections.Generic;
using System.Xml.Linq;


public class Ticket
{
    private string _currentVehicle ;

    public string CurrentValue => _currentVehicle;

    public Ticket()
    {

    }
    public void TakTicket(Queue<int> Ticket, int number )
    {
        Ticket.Enqueue(number);
        Console.WriteLine($"Ticket {number} issued.");   
    }

    public void TicketProcess(Queue<int> Ticket)
    {
        if (Ticket.Count > 0)
        {
            int passedTicket = Ticket.Dequeue();
            Console.WriteLine($"Processing Ticket: {passedTicket}");

            Console.WriteLine("Remaining Tickets: " + (Ticket.Count > 0 ? string.Join(", ", Ticket) : "No more tickets in the queue."));

        }
        else
        {
            Console.WriteLine("No more tickets in the queue.");

        }

    }


}
