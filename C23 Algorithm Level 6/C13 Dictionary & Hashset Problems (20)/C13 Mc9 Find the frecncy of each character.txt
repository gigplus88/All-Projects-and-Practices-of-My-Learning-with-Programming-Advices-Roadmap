using System;

using System.Collections.Generic;
using System.Linq;



class Program
{


    static void Main()
    {
        /*
        Example:
        Input: "hello"
        Output: { 'h': 1, 'e': 1, 'l': 2, 'o': 1 }


         */


        Dictionary<char,int> frequence = new Dictionary<char,int>();

        string Input = "hello";


        /*

        {
            if (!frequence.ContainsKey(c))
            {
                frequence.Add(c, 1);
            }
            else
            {
                frequence[c]++;
            }
        }

         */

        for (int i = 0; i < Input.Length; i++)
        {
            char c = Input[i];
            if (!frequence.ContainsKey(c))
            {
                frequence.Add(c, 1);
            }
            else
            {
                frequence[c]++;
            }
        }

        foreach (var item in frequence)
        {
            Console.Write($"{item.Key}:{item.Value} ");
        }

              


        Console.ReadKey();
    }
}
