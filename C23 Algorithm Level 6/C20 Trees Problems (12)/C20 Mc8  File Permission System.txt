using System;
using System.Collections.Generic;
using System.Security;

public class FileNode
{
    public string Name { get; set; }
    public bool IsFile { get; set; }
    public FileNode Parent { get; set; }
    public List<FileNode> Children { get; set; } = new List<FileNode>();

    private string _permission;

    public enum enType { Directory, File }

    public FileNode(string name, enType fType, string permission = "")
    {
        Name = name;
        IsFile = (fType == enType.File);
        _permission = permission;
    }

    public string Permission
    {
        get
        {
            if (!string.IsNullOrEmpty(_permission))
                return _permission;

            if (Parent != null)
                return Parent.Permission;

            return "No Permission Defined";
        }
        set { _permission = value; }
    }

    public void AddChild(FileNode child)
    {
        child.Parent = this;
        this.Children.Add(child);
    }

    public int GetTotalCount()
    {
        int count = 0;
        foreach (var child in Children)
        {
            count++;
            count += child.GetTotalCount();
        }
        return count;
    }

    public void Print(string  inheritPermission="", string indent = "")
    {
        string typeLabel = IsFile ? "[File]" : "[Dir] ";

        string effectivePermission = _permission == "" ? inheritPermission : _permission; // Get the new permission if it not founded
        Console.WriteLine($"{indent}{Name} | Permissions: {effectivePermission}");

        foreach (var child in Children)
        {
            child.Print(effectivePermission, indent + "  ");
        }
    }
}

class Program
{
    static void Main()
    {
        var root = new FileNode("C:", FileNode.enType.Directory, "rwx");

        var docs = new FileNode("Documents", FileNode.enType.Directory); // سيرث rwx
        var privateFiles = new FileNode("Private", FileNode.enType.Directory, "rw-"); 


        var resume = new FileNode("Resume.pdf", FileNode.enType.File);
        var secret = new FileNode("Pass.txt", FileNode.enType.File);

        root.AddChild(docs);
        root.AddChild(privateFiles);

        docs.AddChild(resume);
        privateFiles.AddChild(secret);

        Console.WriteLine("=== File System Hierarchy ===");
        root.Print();




        Console.WriteLine("\n------------------------------");
        Console.WriteLine($"Total Elements in System: {root.GetTotalCount()}");


        Console.WriteLine("\nUpdating Root permission to 'r--'...");
        root.Permission = "r--";

        root.Print();

        Console.ReadKey();
    }
}
