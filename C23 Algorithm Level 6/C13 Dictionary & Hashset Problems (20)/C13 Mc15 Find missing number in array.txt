    static int FindMissingNum(int[] input)
    {
        Dictionary<int, bool> map = new Dictionary<int, bool>();

        //var res = input.OrderBy(x => x).ToList();

        foreach (var item in input)
        {
            map[item] = true;
        }

        for (int i = 0; i < input.Length; i++)
        {
            if (!map.ContainsKey(i))
                return i;
        }
        return -1;
        
    }
