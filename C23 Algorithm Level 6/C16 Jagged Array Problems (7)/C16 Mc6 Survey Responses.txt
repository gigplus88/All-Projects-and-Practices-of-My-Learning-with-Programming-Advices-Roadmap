
using System;

using System.Collections.Generic;
using System.Linq;
using System.Xml.Linq;

class Program
{
   
 
    static void Main()
    {
        string[][] surveyResponses = new string[3][];

        surveyResponses[0] = new string[] { "Yes", "No", "No" };// Respondent 1
        surveyResponses[1] = new string[] { "Yes", "No", "Yes" };  // Respondent 2
        surveyResponses[2] = new string[] {  "No", "No" };    // Respondent 3





        for (int i = 0; i < surveyResponses.Length; i++)
        {
            Console.Write($"Response {i+1}: ");



            foreach (string item in surveyResponses[i])
            {
                Console.Write( item + " ");
            }

            Console.WriteLine();
        }


        Console.ReadKey();
    }
}
