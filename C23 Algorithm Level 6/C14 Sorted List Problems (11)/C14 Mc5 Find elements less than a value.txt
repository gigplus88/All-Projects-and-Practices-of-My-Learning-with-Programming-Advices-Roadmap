

using System;

using System.Collections.Generic;
using System.Linq;

class Program
{

    static SortedSet<int> RemoveElementsLessThanRange(int[] elements , int Range )
    {
        SortedSet<int> s1 = new SortedSet<int>(elements) ;

        
         s1.RemoveWhere( x => x >= Range) ;
        //return   s1.GetViewBetween(s1.Min , Range-1);
       return s1;
    }

    static IEnumerable<int> ElementsLessThan(int[] elements, int Range)
    {
        SortedSet<int> s1 = new SortedSet<int>(elements);


        return   s1.GetViewBetween(int.MinValue , Range-1);
    }

    static void Main()
    {


        /*
        Example:
        Input: set = [1, 2, 3, 4, 5], value = 4
        Output: [1, 2, 3]
     */




        int[] input = { 1, 2, 4, 3 , 6 };


        

        Console.WriteLine(string.Join(", ", RemoveElementsLessThanRange( input , 4 ))); // Output: 1, 2, 3, 4, 5
        Console.WriteLine(string.Join(", ", ElementsLessThan(input, 4))); // Output: 1, 2, 3, 4, 5


        Console.ReadKey();
