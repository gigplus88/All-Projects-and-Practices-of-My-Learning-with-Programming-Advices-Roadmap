
using System;
using System.Collections.Generic;
using System.Xml.Linq;


public class TrafficSystem
{
    private string _currentVehicle ;

    public string CurrentValue => _currentVehicle;

    public TrafficSystem()
    {

    }
    public void TrafficProcess(Queue<string> Traffic )
    {
        
        if (Traffic.Count > 0)
        {
            string passedVehicle = Traffic.Dequeue();
            Console.WriteLine($"{passedVehicle} has passed.");

            Console.WriteLine("Vehicles waiting: " + (Traffic.Count > 0 ? string.Join(", ", Traffic) : "No vehicles"));

        }
        else
        {
            Console.WriteLine("No vehicles waiting");

            _currentVehicle = "No vehicles waiting";
        }
            
        //Print();
    }


}
