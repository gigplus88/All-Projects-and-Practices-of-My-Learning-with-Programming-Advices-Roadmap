using System;

using System.Collections.Generic;

using System.Threading;
class Program

    static void Main()
    {

        SortedSet< DateTime > items = new SortedSet< DateTime>();

      


        items.Add((  new DateTime(2024, 11, 19, 9, 0, 0) ));

        }
