

using System;

using System.Collections.Generic;
using System.Linq;




class Program
{

    static bool CheckRangesInSet(int[] elements, int[] Range1 , int[] Range2)
    {
        if (elements.Length == 0)
            return false;
           
        SortedSet<int> s1 = new SortedSet<int>(elements);



        bool IsRangeContained(int[] r)
        {
            int min = r.Min();
            int max = r.Max();

            // التحقق: هل أصغر وأكبر قيمة في النطاق موجودتان فعلياً في المجموعة؟
            return s1.Contains(min) && s1.Contains(max);
        }
        //return (Range1.Min() >= MinElements && Range1.Max() <= MaxElements) && (Range2.Min() >= MinElements && Range2.Max() <= MaxElements);
        return IsRangeContained(Range1) && IsRangeContained(Range2);
    }


    static void Main()
    {


        /*
      Example:
        Input: set = [1, 2, 3, 4, 5], value = 3
        Output: 3


     */




        //int[] input = { 1, 2, 4, 3 };
        int[] input = { 1 , 3 , 2 , 4 , 5 ,6,7 };

        int[] r1 = { 1,2 };
        int[] r2 = { 5,9};


        Console.WriteLine("r1 min" + r1.Min());
        Console.WriteLine("r2 max" + r1.Max());


        Console.WriteLine( "r2 min"+ r2.Min());
        Console.WriteLine("r2 max" + r2.Max());


        Console.WriteLine( $" result:{CheckRangesInSet(input, r1 , r2) }  "); // Output: 1, 2, 3, 4, 5



        Console.ReadKey();


    }
}
