//add.dart




import 'package:flutter/material.dart';
import 'package:flutter_pro/sqldb.dart';
import 'package:get/get_core/src/get_main.dart';
import 'package:get/route_manager.dart'; // تأكد أن المسار صحيح لمجلد مشروعك

class AddNote extends StatefulWidget {
  const AddNote({super.key});

  @override
  State<AddNote> createState() => _AddNoteState();
}

class _AddNoteState extends State<AddNote> {
  SqlDb sqlDb = SqlDb();

  final TextEditingController noteController = TextEditingController();
  final TextEditingController title = TextEditingController();
  final TextEditingController color = TextEditingController();

  GlobalKey<FormState> formState = GlobalKey();
  bool isLoading = false;

  @override
  void dispose() {
    noteController.dispose();
    super.dispose();
  }

  Future<void> addNoteToDatabase() async {
    String noteText = noteController.text.trim();
    String noteTitleTxt = title.text.trim();
    String noteColorTxt = color.text.trim();

    if (noteText.isEmpty) {
      ScaffoldMessenger.of(
        context,
      ).showSnackBar(const SnackBar(content: Text('Please Add your note !')));
      return;
    }

    setState(() {
      isLoading = true;
    });

    try {
      int response = await sqlDb.insertData('''
        INSERT INTO notes (note , title , color) 
        VALUES ('$noteText' ,'$noteTitleTxt' , '$noteColorTxt')
        
        ''');

      print("Response ID: $response");

      if (response > 0) {
        if (!mounted) return;
        ScaffoldMessenger.of(context).showSnackBar(
          const SnackBar(content: Text('Note Added Successfully ')),
        );

        Get.offNamed("/homepage");
      }
    } catch (e) {
      print("Error: $e");
      ScaffoldMessenger.of(
        context,
      ).showSnackBar(const SnackBar(content: Text('Error to Add this Note ')));
    } finally {
      if (mounted) {
        setState(() {
          isLoading = false;
        });
      }
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Add Note')),
      body: Padding(
        padding: const EdgeInsets.all(16.0),
        child: Form(
          key: formState,
          child: Column(
            children: [
              //Title
              TextField(
                controller: title,
                //maxLines: 4,
                decoration: const InputDecoration(
                  hintText: 'Enter your Title here...',
                  border: OutlineInputBorder(),
                ),
              ),
              const SizedBox(height: 20),
              TextField(
                controller: noteController,
                maxLines: 4,
                decoration: const InputDecoration(
                  hintText: 'Enter your Note here...',
                  border: OutlineInputBorder(),
                ),
              ),
              const SizedBox(height: 20),
              TextField(
                controller: color,
                //maxLines: 4,
                decoration: const InputDecoration(
                  hintText: 'Enter your color here...',
                  border: OutlineInputBorder(),
                ),
              ),
              const SizedBox(height: 20),

              SizedBox(
                width: double.infinity,
                height: 50,
                child: ElevatedButton(
                  onPressed: isLoading ? null : addNoteToDatabase,
                  child: isLoading
                      ? const CircularProgressIndicator(color: Colors.white)
                      : const Text(
                          'Save Note ',
                          style: TextStyle(fontSize: 18),
                        ),
                ),
              ),
            ],
          ),
        ),
      ),
    );
  }
}

import 'package:flutter/material.dart';
import 'package:flutter_pro/sqldb.dart';
import 'package:get/get_core/src/get_main.dart';
import 'package:get/route_manager.dart';

class EditNote extends StatefulWidget {
  final int id;
  final String note;
  final String title;
  final String color;
  EditNote({
    super.key,
    required this.id,
    required this.note,
    required this.title,
    required this.color,
  });

  @override
  State<EditNote> createState() => _EditNoteState();
}

class _EditNoteState extends State<EditNote> {
  SqlDb sqlDb = SqlDb();

  final TextEditingController noteController = TextEditingController();
  final TextEditingController title = TextEditingController();
  final TextEditingController color = TextEditingController();

  GlobalKey<FormState> formState = GlobalKey();
  bool isLoading = false;
  initialData() {
    noteController.text = widget.note;
    title.text = widget.title;
    color.text = widget.color;
  }

  @override
  void initState() {
    initialData();
    super.initState();
  }

  @override
  void dispose() {
    noteController.dispose();
    super.dispose();
  }

  Future<void> editNote() async {
    String noteText = noteController.text.trim();
    String noteTitleTxt = title.text.trim();
    String noteColorTxt = color.text.trim();
    int id = widget.id;
    if (noteText.isEmpty) {
      ScaffoldMessenger.of(
        context,
      ).showSnackBar(const SnackBar(content: Text('Please edit your note !')));
      return;
    }

    setState(() {
      isLoading = true;
    });

    try {
      int response = await sqlDb.updateData('''
        UPDATE notes Set  
        note = '$noteText' ,
        title = '$noteTitleTxt' ,
        color = '$noteColorTxt'
        Where id = '$id'
        ''');

      print("Response ID: $response");

      if (response > 0) {
        if (!mounted) return;
        ScaffoldMessenger.of(context).showSnackBar(
          const SnackBar(content: Text('Note edited Successfully ')),
        );

        Get.offNamed("/homepage");
      }
    } catch (e) {
      print("Error: $e");
      ScaffoldMessenger.of(
        context,
      ).showSnackBar(const SnackBar(content: Text('Error to edit this Note ')));
    } finally {
      if (mounted) {
        setState(() {
          isLoading = false;
        });
      }
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Edit Note')),
      body: Padding(
        padding: const EdgeInsets.all(16.0),
        child: Form(
          key: formState,
          child: Column(
            children: [
              //Title
              TextField(
                controller: title,
                //maxLines: 4,
                decoration: const InputDecoration(
                  hintText: 'Enter your Title here...',
                  border: OutlineInputBorder(),
                ),
              ),
              const SizedBox(height: 20),
              TextField(
                controller: noteController,
                maxLines: 4,
                decoration: const InputDecoration(
                  hintText: 'Enter your Note here...',
                  border: OutlineInputBorder(),
                ),
              ),
              const SizedBox(height: 20),
              TextField(
                controller: color,
                //maxLines: 4,
                decoration: const InputDecoration(
                  hintText: 'Enter your color here...',
                  border: OutlineInputBorder(),
                ),
              ),
              const SizedBox(height: 20),

              SizedBox(
                width: double.infinity,
                height: 50,
                child: ElevatedButton(
                  onPressed: isLoading ? null : editNote,
                  child: isLoading
                      ? const CircularProgressIndicator(color: Colors.white)
                      : const Text(
                          'Save Note ',
                          style: TextStyle(fontSize: 18),
                        ),
                ),
              ),
            ],
          ),
        ),
      ),
    );
  }
}
