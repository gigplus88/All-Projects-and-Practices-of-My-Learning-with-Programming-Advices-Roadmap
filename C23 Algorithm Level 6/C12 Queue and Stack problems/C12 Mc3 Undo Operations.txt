using System;
using System.Collections.Generic;

public class UndoCalculator
{
    private double _currentValue = 0;
    private Stack<double> _history = new Stack<double>();

    public double CurrentValue
    { 
        get { return _currentValue; }
      
        set 
        {
            _currentValue = value;
        } 
    }
   // public double CurrentValue => _currentValue;

    public void Add(double value)
    {
        _history.Push(_currentValue);
        _currentValue += value;
        Console.WriteLine($"Add value :{value} , with result: {_currentValue}");

    }
    public void Substract(double value)
    {
        _history.Push(_currentValue);
        _currentValue -= value;
        Console.WriteLine($"substract value :{value} , with result: {_currentValue}");

    }
     public void Undo()
    {
        if (_history.Count >0)
        {
            _currentValue = _history.Pop();
            Console.WriteLine($"Old result is: {_currentValue}");
        }
      
    }

}
