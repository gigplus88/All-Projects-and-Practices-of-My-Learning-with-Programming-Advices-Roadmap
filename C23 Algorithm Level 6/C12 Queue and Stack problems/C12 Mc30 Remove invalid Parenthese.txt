    static string RemoveParenthese(string value )
    {
        value = value.Trim();
        Stack<int> stack = new Stack<int>();

        HashSet<int> invalidIndexes = new HashSet<int>();


        int[] FaultIndex = new int[value.Length];
        for (int i = 0; i < value.Length; i++)
        {
            if (value[i] == '(')
            {
                stack.Push(i);
                //Rightcounter++;
            }
            else if (value[i] == ')')
            {
                // if parenthese is in the final of the input
                if (stack.Count ==0)
                {
                    invalidIndexes.Add(i);
                }

                else
                {
                    stack.Pop();
                }
                // Leftcounter++;
            }
        }
        //if (stack.Count > 0)
        //{
        //    InvalidIndecies[i] = i;
        //}


        //To handle if the first num of parentheses is great of the final
        while (stack.Count > 0)
        {
            invalidIndexes.Add(stack.Pop());
        }

        StringBuilder result = new StringBuilder();

        for (int i = 0; i < value.Length; i++)
        {
            if (!invalidIndexes.Contains(i))
            {
                result.Append(value[i]);
            }
               
        }

        //int Counter = stack2.Count;

        //for (int i = 0; i < Counter; i++)
        //{
        //    stack2.Push(stack2.Pop());

        //}

        return result.ToString();
    }
