import { useState } from "react";
import "./index.css";
import Modal from "./Modal";

function MyForm() {
  const [formInputs, setFormInputs] = useState({
    name: "",
    phone: "",
    isEmployee: false,
    age: "",
    salary: "Less than 5000",
  });

  // setFormInputs for updating the form inputs
  // formInputs for storing the values of form inputs
  const [error, setError] = useState("");

  function handleChange(event) {
    const { name, value, type, checked } = event.target;

    setFormInputs({
      ...formInputs,
      [name]: type === "checkbox" ? checked : value,
    });
  }
  function showMessage(msg) {
    setError(msg);
    //setTimeout(() => setError(""), 3000);
  }
  function handleSubmit(event) {
    event.preventDefault();
    //event.stopPropagation();

    const phoneNumber = formInputs.phone;
    const age = Number(formInputs.age);

    //const {age , phoneNumber} = formInputs;
    setError("");
    if (!/^\d{10}$/.test(phoneNumber) ) {
      showMessage(
        "Please enter a valid phone number (10 digits) and not words",
      );
      return;
    }

    if (!age || age < 18 || age > 100) {
      showMessage("Please enter a valid age (18 - 100)");
      return;
    }
    showMessage("Form submitted successfully!");

    console.log("Form Data:");
    console.log(formInputs);
  }

  //To close Modal when user clicks outside of it
  function handleDivClick() {
    if (error) {
      setError("");
    }
      
  }
  const btnIsDisabled =
    !formInputs.name || !formInputs.phone || !formInputs.age;

  return (
    <div
      style={{
        padding: "20px",
        border: "1px solid black",
        borderRadius: "15px",
        margin: "20px",
        width: "800px",
      }}
      onClick={handleDivClick}>
      <h1>Requesting a loan</h1>

      <hr />

      <form
        onSubmit={handleSubmit}
        style={{ display: "flex", flexDirection: "column", gap: "10px" }}>
        <label>Name</label>
        <input
          type="text"
          name="name"
          value={formInputs.name}
          onChange={handleChange}
          className="inputStyle"
        />

        <hr />

        <label>Phone Number</label>
        <input
          type="text"
          name="phone"
          value={formInputs.phone}
          onChange={handleChange}
          className="inputStyle"
        />

        <hr />

        <label>Age</label>
        <input
          type="number"
          name="age"
          value={formInputs.age}
          onChange={handleChange}
          className="inputStyle"
        />

        <hr />

        <label>Are you an Employee</label>
        <input
          type="checkbox"
          name="isEmployee"
          checked={formInputs.isEmployee}
          onChange={handleChange}
        />

        <hr />

        <label>Salary</label>
        <select
          name="salary"
          value={formInputs.salary}
          onChange={handleChange}
          className="inputStyle">
          <option>Less than 5000</option>
          <option>5000 - 10000</option>
          <option>More than 10000</option>
        </select>

        <hr />

        <button
          className={btnIsDisabled ? "disabledButton" : "enabledButton"}
          disabled={btnIsDisabled}
          type="submit">
          Submit
        </button>
      </form>

      <Modal isVisible={!!error} message={error} />
    </div>
  );
  /*{error && (<Modal
        title="Form Status"
        content={error}
        onClose={() => setError("")}></Modal>
      )}*/
}

export default MyForm;

export default function Modal({ isVisible , message = null }) {
  const isSuccess = message?.includes("successfully");

   if (isVisible) {
  return (
    <div className="modalOverlay">
      <div className="modalContent" onClick={(e) => e.stopPropagation()}>
        <p style={{ color: isSuccess ? "green" : "red" }}>
          {isSuccess ? "The form is submitted successfully." : message}
        </p>
      </div>
    </div>
  );}
   else { return( <></>)}
}
