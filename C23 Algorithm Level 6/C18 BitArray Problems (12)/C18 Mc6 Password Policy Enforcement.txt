using System;
using System.Collections;
using System.Collections.Generic;
using System.Diagnostics;
using System.Net.Cache;
using System.Xml.Linq;
class Program
{

    static void BuildPassword(string password , BitArray strengh)
    {
        if (password == null) Console.WriteLine("password required");

        foreach (var c in password)
        {
            if (char.IsUpper(c)) strengh[0] = true;
            else if (char.IsLower(c)) strengh[1] = true;
            else if (char.IsDigit(c)) strengh[2] = true;
            else if (char.IsSymbol(c)|| char.IsPunctuation(c)) strengh[3] = true;
        }
    }

    static void DisplayPasswordStatus( BitArray strengh)
    {
        string[] criteria = { "Upper letter", "lower letter", "Digit", "symbol" };
        int score = 0;
        for (int i = 0; i < strengh.Count; i++)
        {
            if (strengh[i]) score++;

            Console.WriteLine($"{criteria[i]} , {(strengh[i] ? "good" : "bad")} ");
        }
        Console.WriteLine($"Score: {score}/4");
        if (score == 4) Console.WriteLine("Strong password 💪");
        else Console.WriteLine("Bad password Try to enter Uppercase lowercase digit and symbol in the same password.");
    }
    static void Main()
    {


         BitArray strengh = new BitArray(4, false); // All lights off


        Console.WriteLine("Enter your password");
        string Password = Console.ReadLine();

        BuildPassword(Password, strengh);

        DisplayPasswordStatus(strengh);





        //Teacher solution

        // Output: Password Strength: True, True, True, True

    }
}
