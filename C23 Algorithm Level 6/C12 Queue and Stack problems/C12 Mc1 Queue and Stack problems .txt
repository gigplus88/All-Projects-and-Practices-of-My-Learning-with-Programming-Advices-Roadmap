        Implement Browser Back Button
        Problem: Use a stack to implement a browser's back button functionality.
    */

    public class clsBrowse
    {
        private Stack<string> goBack;
        private Stack<string> goForward;
        private string Current;

        public clsBrowse(string Site)
        {
            Current = Site;
            this.goBack =  new Stack<string>();
            this.goForward = new Stack<string>();
        }

        public void GoBack()
        {
            if (goBack.Count > 0)
            {
                Current = this.goBack.Pop();
                goForward.Push(this.Current);
            }
         
            this.Print();
            //return this.Current;
        }
        public void GoForward()
        {
            this.Current = this.goForward.Pop();
            this.goBack.Push(this.Current);
            this.Print();
            //return this.Current;
        }

        public string GetCurrent()
        {
            if (this.goBack.Count >0)
            {
                return this.goBack.Peek();
            }
            return this.Current;
        }
        public void GoToWebSite( string Site)
        {
            this.goBack.Push(this.Current); // I preserve current page 
            this.Current = Site;

            goForward.Clear();

            this.Print();
           
            //return true;
        }
        public void Print() { Console.WriteLine("Connected Now To: " + this.Current?? "Empty history"); }
    }
