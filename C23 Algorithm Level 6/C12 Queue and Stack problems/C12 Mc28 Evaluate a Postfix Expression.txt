using System;
using System.Collections;
using System.Collections.Generic;
using System.Diagnostics;
using System.Linq;
using System.Management.Instrumentation;
using System.Text;
using static Guna.UI2.Native.WinApi;


class Program
{
   static int Operation(char Op ,int FirstVal , int LastVal)
    {
        int Res = 0;
        switch (Op)
        {
            case '+':
                Res = FirstVal + LastVal;
                break;

            case '-':
                Res = FirstVal - LastVal;

                break;

            case '*':
                Res = FirstVal * LastVal;

                break;

            case '/':
                Res = FirstVal / LastVal;

                break;

        }
        return Res;
    }
   
    static int EvaluatePostfix(string value )
    {

        Stack<int> stack = new Stack<int>();

        
        int lastValue = 0 , FirstValue = 0;
        int Res = 0;
        foreach (var item in value)
        {
            if (int.TryParse(item.ToString(), out int number) )
            {
                stack.Push(number);
            }
            else
            {
                lastValue = stack.Pop();
                FirstValue = stack.Pop();

                Res = Operation(item.ToString(), FirstValue , lastValue);

                stack.Push(Res );
            }
                
        }
        return stack.Pop() ;
    }


    static void Main()
    {


        string value = "231*+9-";
        

        Console.WriteLine("check is value palindrome : " + EvaluatePostfix(value));



        Console.ReadKey();
    }
}
