using System;
using static LoggingService;

public interface ILogging
{
     void Log(string message);
}
public class LoggingService
{
    private ILogging _Logging;
   // public enum enLoggingType { ToFile, ToEventLog, ToDatabase }
   public LoggingService(ILogging notification)
   {
        _Logging = notification;
        
   }
   
    public void SendNotification( string message = "")
    {

        _Logging.Log(message);

    }


}
public class LogToEventLog:ILogging
{
    public string Message { get; set; }

    public  void  Log(string message )
    {

        Console.WriteLine($"\nLog to EventLog: {message}");
    }
}
public class LogToFile: ILogging
{
    public void Log(string message)
    {
        Console.WriteLine($"\nLog to file: {message}");
    }
}
public class LogToDatabase: ILogging
{
    public string Message { get; set; }

    public void Log(string message)
    {
        Console.WriteLine($"\nLog to Database: {message}");
    }
}
class Program
{
    static void Main()
    {
        LoggingService LoggingService = new LoggingService( new LogToFile()); 

        LoggingService.SendNotification("Error Occured line xxx.");


        LoggingService = new LoggingService(new LogToEventLog()); 

        LoggingService.SendNotification("Error Occured line xxx.");


        LoggingService = new LoggingService(new LogToDatabase()); 

        LoggingService.SendNotification("Error Occured line xxx.");



        Console.ReadKey();

    }
}
