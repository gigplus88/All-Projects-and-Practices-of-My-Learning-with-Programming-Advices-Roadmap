using System;
using System.Collections;
using System.Collections.Generic;
using System.Globalization;
using System.Linq;


class WebServer
{

    // My Solution

    static Queue<int> InterleaveQueueElements(Queue<int> queue)
    {
	if (queue.Count % 2 != 0)
    	   throw new ArgumentException("queue items must be odd");
        Queue<int> FirstHalfQueue = new Queue<int>(); //queue.Take(queue.Count / 2)

        int HalfCount = queue.Count / 2;


        for (int i = 0; i < HalfCount; i++)
        {

            FirstHalfQueue.Enqueue(queue.Dequeue());
        }
        // Other items are in the queue


        while (FirstHalfQueue.Count > 0)
        {

            queue.Enqueue(FirstHalfQueue.Dequeue());

            queue.Enqueue(queue.Dequeue());
        }
        // this will generate a queue from list
        return queue;
    }



    static void Main()
    {

        // Teacher Solution

        Queue<int> queue = new Queue<int>(new[] { 1, 2, 3, 4, 5, 6 });




        Console.WriteLine(string.Join(",", InterleaveQueueElements(queue)));







    }

}
