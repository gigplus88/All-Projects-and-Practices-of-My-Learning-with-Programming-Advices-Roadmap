// Violation of ISP: An interface that forces clients to implement methods they don't need
using System;
using System.Drawing;


public interface IPrint
{
    void Print(string content);
}
public interface IScan
{
    void Scan();
}
public interface IFax
{
    void Fax();
}
public interface IDesign
{
    void Design();
}
public class BasicPrinter : IPrint
{
    public void Print(string content)
    {
        Console.WriteLine($"Printing: {content}");
    }
}


public class AdvancedPrinter :IPrint,IFax,IScan,IDesign
{
    public void Print(string content)
    {
        Console.WriteLine($"Printing: {content}");
    }

    public void Scan()
    {
        Console.WriteLine($"Scanning..");
    }

    public void Fax()
    {
        Console.WriteLine($"Faxing...");
    }
    public void Design()
    {
        Console.WriteLine($"Designing...");
    }
}

public class Program
{
    public static void Main()
    {
        IPrint basicPrinter = new BasicPrinter();
        basicPrinter.Print("Hi, My Name is Abu-Hahdoud");
        // basicPrinter.Scan();

        //basicPrinter.Fax();


        // Simple Printer have not Fax and scan implementation
        BasicPrinter basicPrinter1 = new BasicPrinter();
        basicPrinter1.Print("go");
        //basicPrinter1.Fax();
        //basicPrinter1.Scan();



        AdvancedPrinter advancedPrinter = new AdvancedPrinter();
        advancedPrinter.Print("Hi, My Name is Abu-Hahdoud");
        advancedPrinter.Scan();
        advancedPrinter.Fax();

        Console.ReadKey();

    }
}
