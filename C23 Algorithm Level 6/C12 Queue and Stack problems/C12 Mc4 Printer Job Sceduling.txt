
using System;
using System.Collections.Generic;

public class UndoCalculator
{
    private double _currentValue = 0;
    private Stack<double> _history = new Stack<double>();

    public double CurrentValue
    { 
        get { return _currentValue; }
      
        set 
        {
            _currentValue = value;
        } 
    }
   // public double CurrentValue => _currentValue;

    public void Add(double value)
    {
        _history.Push(_currentValue);
        _currentValue += value;
        Console.WriteLine($"Add value :{value} , with result: {_currentValue}");

    }
    public void Substract(double value)
    {
        _history.Push(_currentValue);
        _currentValue -= value;
        Console.WriteLine($"substract value :{value} , with result: {_currentValue}");

    }
     public void Undo()
    {
        if (_history.Count >0)
        {
            _currentValue = _history.Pop();
            Console.WriteLine($"Old result is: {_currentValue}");
        }
      
    }

}

    static void Main()
    {
        Queue<string> jobs = new Queue<string>();

        jobs.Enqueue("job1");
        jobs.Enqueue("job2");
        jobs.Enqueue("job3");
        jobs.Enqueue("job4");
        jobs.Enqueue("job5");
        jobs.Enqueue("job6");
        int i = 0;
        foreach (var item in jobs)
        {
            i++;
            Console.WriteLine($"Jobs {i}:" + item);
 
        }
        if (jobs.Count >0)
        {
            Console.WriteLine("Processing: " + jobs.Dequeue()); // Output: Processing: Document1
            Console.WriteLine("Next Job: " + jobs.Peek()); // Output: Next Job: Document2
            Console.WriteLine("Jobs by levels:" + jobs.Dequeue());
        }





        Console.ReadKey();
    }
}
