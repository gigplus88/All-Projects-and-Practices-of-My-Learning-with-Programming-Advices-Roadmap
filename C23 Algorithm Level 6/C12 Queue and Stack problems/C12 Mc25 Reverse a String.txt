using System;
using System.Collections;
using System.Collections.Generic;
using System.Linq;
using System.Management.Instrumentation;
using System.Text;


class Program
{
   
    static string ReverseString(string value )
    {
        Stack<char> stack = new Stack<char>(value);
        StringBuilder res = new StringBuilder();

        while (stack.Count > 0)
        {
            res.Append(stack.Pop());
        }
        return res.ToString();

        //return new string(value.Reverse().ToArray() );


    }


    static void Main()
    {

        /*
            Problem: Reverse a given string using a stack.

            Input: "hello"
            Output: "olleh"
         */
      


        Console.WriteLine("Middle is : " + ReverseString( "hello"));



        Console.ReadKey();
    }
}
