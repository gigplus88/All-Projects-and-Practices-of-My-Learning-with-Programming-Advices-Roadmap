    // Priority by priority num then Desc
    class Task :IComparable<Task>
    {
        public int Priority { get; set; }
        public string Desc { get; set; }
        public Task(int priority , string Desc) 
        {
            this.Priority = priority;
            this.Desc = Desc;
        }

        public int CompareTo(Task other) 
        {
            int res = Priority.CompareTo(other.Priority);

            if(res ==0) res = Desc.CompareTo(other.Desc);

            return res;
        }
    }
