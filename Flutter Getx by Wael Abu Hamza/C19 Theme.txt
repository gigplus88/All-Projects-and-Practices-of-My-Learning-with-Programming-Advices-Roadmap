//ThemeController 


import 'package:flutter/material.dart';
import 'package:flutter_pro/main.dart';
import 'package:get/get.dart';

class ThemeController extends GetxController {
  String? themeName;
  ThemeData? myTheme;

  @override
  void onInit() {
    if (sharedPreferences!.get("theme") == null)
      myTheme = Themes.customLightTheme;
    else
      myTheme = Themes.customDarkTheme;

    super.onInit();
  }
}

          ],
        ),
      ),

  }
}








//PageTwoController 



import 'package:get/get.dart';

class PageTwoController extends GetxController {
  String? name;

  String? previousRoute;
  String? currentRoute;

  @override
  void onInit() {
    if (Get.arguments != null) {
      name = Get.arguments['name'] ?? "";
      previousRoute = Get.previousRoute;
      currentRoute = Get.routing.current;
    }

    super.onInit();
  }
}
