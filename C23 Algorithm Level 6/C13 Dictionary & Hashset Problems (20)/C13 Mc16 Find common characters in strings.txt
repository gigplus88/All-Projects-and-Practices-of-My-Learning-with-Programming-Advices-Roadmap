//teacher solution

using System;
using System.Collections.Generic;

class Program
{
    static List<string> CommonChars(string[] words)
    {
        int[] minFreq = new int[26];
        Array.Fill(minFreq, int.MaxValue); //Get to all minFreq int.MaxValue


        foreach (string word in words)
        {
            int[] charFreq = new int[26];
            foreach (char c in word) // freq of all letter in word
            {
                charFreq[c - 'a']++;  //'a' is one
            }


            for (int i = 0; i < 26; i++) // store min frequencies of all letters by comparing it with begin word letter freq 
            {
                minFreq[i] = Math.Min(minFreq[i], charFreq[i]); // ex Math.Min( 2147483647 , 2)  2147483647 for minFreq , and 2 for number of 'l' frequencies
            }
        
        }

        // Print frequencies letters
        List<string> result = new List<string>();
        for (int i = 0; i < 26; i++)
        {
            for (int j = 0; j < minFreq[i]; j++) // ex: l has 2 frequencies
            {
                result.Add( ((char)(i + 'a')).ToString());
            }
        }

        return result;
    }

    static void Main()
    {
        var result = CommonChars(new[] { "bella", "label", "roller" });
        Console.WriteLine(string.Join(", ", result)); // Output: e, l, l
        Console.ReadKey();

    }
}
