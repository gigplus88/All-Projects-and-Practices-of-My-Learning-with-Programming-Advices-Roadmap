using System;
using System.Collections;
using System.Collections.Generic;
using System.Diagnostics;
using System.Net.Cache;
using System.Xml.Linq;
class Program
{

    static void Main()
    {


         BitArray schedule = new BitArray(7, true); // All lights off

        schedule[5] = false;  // day 6 is free
        schedule[2] = false;  // day 3 is free


        Console.WriteLine($"Day 6 is: { (schedule[5] ? "not free" : "free")}"); // Output: True
        Console.WriteLine($"Day 2 is: {(schedule[1] ? "not free" : "free")}"); // Output: True



        Console.WriteLine("Free Days:");
        for (int i = 0; i < 7; i++)
        {
            if (!schedule[i]) Console.WriteLine($"Day {i + 1}");
        }


        Console.ReadKey();
    }
}
