using System;
using System.Collections;
using System.Collections.Generic;
using System.Linq;
using System.Management.Instrumentation;
using System.Text;
using static Guna.UI2.Native.WinApi;


class Program
{
   
    static bool CheckPalindrome(string value )
    {

        Stack<char> stack = new Stack<char>(value);

        int Count = stack.Count;
        for (int i = 0; i < Count; i++)
        {
            if( stack.Pop() != value[i] )
              return false;
        }
        return true;
    }


    static void Main()
    {


        string value = "madam" , val = "hello";
        

        Console.WriteLine("check is value palindrome : " + CheckPalindrome(value));
        Console.WriteLine("check is value palindrome : " + CheckPalindrome(val));



        Console.ReadKey();
    }
}
