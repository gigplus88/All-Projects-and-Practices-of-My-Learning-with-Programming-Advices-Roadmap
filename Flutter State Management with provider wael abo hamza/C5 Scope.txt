import 'package:flutter/material.dart';
import 'package:provider/provider.dart';

class Test extends StatefulWidget {
  const Test({super.key});

  @override
  State<Test> createState() => _TestState();
}

class _TestState extends State<Test> {
  @override
  void initState() {
    super.initState();
  }

  @override
  Widget build(BuildContext context) {
    return ChangeNotifierProvider(
      create: (context) => Model(),
      child: Scaffold(
        appBar: AppBar(title: const Text("Add Category")),
        body: Container(
          child: Column(
            mainAxisAlignment: MainAxisAlignment.center,
            children: [
              Selector<Model, String>(
                selector: (context, provname1) => provname1.showone,
                builder: (context, model, child) {
                  print("rebuild text 1");
                  return Center(
                    child: Text(
                      "${model}",
                      style: const TextStyle(fontSize: 24),
                    ),
                  );
                },
              ),
              const SizedBox(height: 10),
              Consumer<Model>(
                builder: (context, model, child) {
                  return MaterialButton(
                    color: Colors.blue,
                    textColor: Colors.white,
                    onPressed: () {
                      model.changeName();
                      print("rebuild button 1");
                    },
                    child: const Text("Do something1"),
                  );
                },
              ),

              Selector<Model, String>(
                selector: (context, provname1) => provname1.showtwo,
                builder: (context, model, child) {
                  print("rebuild text 2");
                  return Center(
                    child: Text(
                      "${model}",
                      style: const TextStyle(fontSize: 24),
                    ),
                  );
                },
              ),
              const SizedBox(height: 10),
              Consumer<Model>(
                builder: (context, model, child) {
                  return MaterialButton(
                    color: Colors.blue,
                    textColor: Colors.white,
                    onPressed: () {
                      model.changeName2();
                      print("rebuild button 2");
                    },
                    child: const Text("Do something2"),
                  );
                },
              ),
            ],
          ),
        ),
      ),
    );
  }
}

class Model extends ChangeNotifier {
  String name1 = "Welcome";
  String name2 = "Welcome2";

  get showone => name1;
  get showtwo => name2;

  void changeName() {
    name1 = "Ayoub";
    notifyListeners();
  }

  void changeName2() {
    name2 = "Ayoub2";
    notifyListeners();
  }
}
