import "./App.css";

import { Routes, Route, Link } from "react-router-dom";
import Hello from "./NewPro/Hello";
import Home from "./NewPro/Home";
import PostList from "./NewPro/PostList";
import Post from "./NewPro/Post";
import { PostContext } from "./Contexts/PostContext";
import NotFound from "./NewPro/NotFound";
import DeletePost from "./NewPro/DeletePost";
import NewPost from "./NewPro/NewPost";
import PostLayout from "./NewPro/PostLayout";
function App() {
  
const postData = [
  {
    id: 1,
    title: "Post Title 1",
    content:
      "Lorem ipsum dolor sit amet, consectetur adipiscing elit. .",
  },
   {
    id: 2,
    title: "Post Title 2",
    content:
      "Lorem ipsum dolor sit amet, consectetur adipiscing elit. .",
  },
   {

    title: "Post Title 3",

  },
];

      </Link>

      <Link to="/posts">
        <button>Posts</button>
      </Link>

       <Link to="/about">
        <button>About</button>
      </Link>

       <Link to="/contact">
        <button>Contact</button>
      </Link>

      <Link to="/Hello">
        <button>Hello</button>
      </Link>

    

      <Routes>
        <Route path="/Hello" element={<Hello />} />

export default App;
