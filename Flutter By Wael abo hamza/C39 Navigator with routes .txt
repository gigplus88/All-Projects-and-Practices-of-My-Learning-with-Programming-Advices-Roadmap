 import 'package:flutter/material.dart';
 import 'package:flutter_pro/homepage.dart';
 import 'package:flutter_pro/aboutus.dart';


 class ContactPage extends StatelessWidget {
   const ContactPage({super.key});

   @override
   Widget build(BuildContext context) {
     return Scaffold(
       appBar: AppBar(
         title: const Text("Contact Us"),
       ),
       body: ListView(
        children:[
             Center(
           child: Text("Welcome to the Contact Us Page"),
         ),
          Container(
            color: Colors.red[200],
            margin: const EdgeInsets.symmetric(horizontal: 20),
            
            child: MaterialButton(
              
              onPressed: () {
                Navigator.push(
                  context,
                  MaterialPageRoute(builder: (context) => const HomePage()),
                );
              },
              child: const Text("Go to Home Page with push"),
            ),
          ),
          Container(
            color: Colors.red[200],
            margin: const EdgeInsets.all(5),
            
            child: MaterialButton(
              
              onPressed: () {
                Navigator.push(
                  context,
                  MaterialPageRoute(builder: (context) => const AboutUsPage()),
                );
              },
              child: const Text("Go to About Us Page with push"),
            ),
          ),

           Container(
            color: Colors.red[200],
            margin: const EdgeInsets.all(5),
            
            child: MaterialButton(
              
              onPressed: () {
                Navigator.pushAndRemoveUntil(
                  context,
                  MaterialPageRoute(builder: (context) => const HomePage()),
                  (route) => false,
                );
              },
              child: const Text("Go to Home Page with pushAndRemoveUntil "),
            ),
          ),
           Container(
            color: Colors.red[200],
            margin: const EdgeInsets.all(5),
            
            child: MaterialButton(
              
              onPressed: () {
                Navigator.pushReplacementNamed(
                  context,
                  '/home',
                );
              },
              child: const Text("Go to Home Page with pushAndRemoveUntil "),
            ),
          ),
        ]
          
       ),
     );
   }
 }
