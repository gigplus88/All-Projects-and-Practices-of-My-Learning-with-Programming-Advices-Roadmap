using System;
using System.Collections.Generic;
using System.Linq;


class WebServer
{
    static Queue<int> ReverseQueue(Queue<int> queue)
    {
        Stack<int> stack = new Stack<int>();
        while (queue.Count > 0)
        {
            stack.Push(queue.Dequeue());
        }
        while (stack.Count > 0)
        {
            queue.Enqueue(stack.Pop());
        }
        return queue;
    }

    static void Main()
    {


        //  Queue<int> queue = new Queue<int>(new[] { 1, 2, 3, 4, 5 });


        int[] queue = { 1, 2, 3, 4, 5 };


        Queue<int> QueueForReverse = new Queue<int>(queue);

        Stack<int> Reverse = new Stack<int>();

        //Reversing with Linq Libraray
        Console.WriteLine($"Output Queue: [{string.Join(",", QueueForReverse.Reverse())}]");


        for (int i = 0; i < queue.Length; i++)
        {
            Reverse.Push(QueueForReverse.Dequeue());
        }


        Console.Write($"Output Queue:[");
        while (Reverse.Count>0)
        {
            int current = Reverse.Pop();
            Console.Write($"{current}");
            Console.Write(",");
        }

        Console.WriteLine($"]");







        Console.ReadKey();
    }
